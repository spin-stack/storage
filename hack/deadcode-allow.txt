# Unreachable-from-a-binary symbols that are not liabilities, each with the reason.
#
# Read hack/deadcode.sh first: it explains what "unreachable" means here (no path from
# ./cmd/... or integration/guestinit), and the two things the analysis cannot see.
#
# Format: `<symbol>  # why`, or `package <import path>  # why`. The reason is not
# decoration — `task deadcode` fails on an entry that has none, and fails on an entry that
# no longer matches a finding. Both rules exist because the list this replaced was a set of
# names with the decisions missing, and it went stale twice without anyone noticing.
#
# What belongs here: an affordance whose only legitimate caller is a test, a DST scenario
# or an integration lane, because those are not binaries and never will be. What does NOT
# belong here: a component that was written for production and never wired up. That is the
# CloneCrossHost shape, and CLAUDE.md's answer to it is deletion, not an entry in a file.

# --- the Agent's read-only accessors ---------------------------------------------------
# A binary never asks the Loop what it thinks; it asks the Control Plane. These four
# accessors exist so a test can observe the lease/desired-state/fencing decisions the loop
# already made, instead of asserting on a field it set itself, which CLAUDE.md forbids.
#
# NewVolumeSet and its three methods left this list on 2026-08-22 by becoming *reachable*:
# they were the stand-in "the Agent runs against when there is no data path", and with the
# local block engine withdrawn there is no other VolumeSource — cmd/volume-agent builds one.
internal/agent.Loop.LeaseValid   # observed by loop_test/spine_test; production reads the lease through the loop's own paths
internal/agent.Loop.Desired      # same: the desired state a test compares against what the loop then did
internal/agent.Loop.HostState    # same
internal/agent.Loop.Fenced       # same

# --- Control-Plane election affordances ------------------------------------------------
internal/controlplane.ParseTermClaim            # the inverse of the claim encoder, so election_test can assert on what was written rather than on the writer's argument
internal/controlplane.Elector.HighestClaimedTerm # read by election_test and by the pg integration lane to prove a zombie CP's term never wins

# --- deterministic ids (INV-22) ---------------------------------------------------------
internal/ids.NewAt  # CLAUDE.md names it: ids.New() in production, ids.NewAt(ms, r) for DST's reproducible ids. A binary using it would be a bug.

# --- the lease manager ------------------------------------------------------------------
# The Agent's loop drives the clock-explicit variants (GrantAt/RenewAt); these two are the
# now-relative spellings the tests and the vhost lane use.
internal/lease.Manager.Grant  # used by internal/lease's own tests and by the DST harness's planted-bug proof; loop.go calls GrantAt

# --- lifecycle enum listings ------------------------------------------------------------
# Each returns every value of one enum. Their reason for existing is that a test can walk
# the whole set: lifecycle_test round-trips every value through Parse/String, and
# pg_integration_test asserts the database's CHECK constraint accepts exactly these — which
# is how a state added in Go but not in schema.sql gets caught.
internal/lifecycle.HostStates       # walked by lifecycle_test and by pg_integration_test against the column's CHECK
internal/lifecycle.CordonReasons    # same
internal/lifecycle.VolumeStates     # same
internal/lifecycle.Refusals         # walked by cpserver/refusal_test against every VolumeRefusal the wire declares, so a value can never exist on one side only
internal/lifecycle.RefusalNames     # compared with volumes.refusal's CHECK list by schema/refusal_test, the same drift the three above are guarded from

# --- the observability test provider ----------------------------------------------------
# obs.NewTestProvider is why every metric in this repository is proven by reading the
# collected value rather than by checking that a recorder was called. It cannot live in a
# _test.go file: internal/agent, internal/wal and internal/obs's own tests all import it.
internal/obs.NewTestProvider          # the in-memory meter provider tests collect from; production wires the OTLP one
internal/obs.Provider.CollectedMetrics # reading it is how a test asserts on what left the process
internal/obs.Provider.GaugeValues      # same, for gauges
internal/obs.Provider.GaugeSeries      # same, for a gauge family whose label set varies — chain_depth has one series per volume and a test has to see them apart
internal/obs.Metrics.Len               # only TestMetricsRegistration calls it, and that is what it is for: every Catalog() entry must have become a live instrument, and counting them is the only way to see one that did not register

# --- packages no binary imports (the coarse pass) -------------------------------------------
package internal/dst        # the deterministic simulation harness. It is the test lane, so a binary importing it would be the defect.
package internal/metadata/sim   # the in-memory metadata.Store DST runs against; binaries take metadata/pg
package internal/metadata/metadatatest # the shared metadata.Store contract both implementations run
package internal/simio/sim  # the simulated clock/disk/network/objectstore; the counterpart of simio/real, which binaries do take
package internal/simio      # the interface package's own directory (docs and shared declarations); binaries import its children
package internal/simio/objectstore/storetest # the shared objectstore contract every store implementation runs
package internal/schema     # schema.sql plus its embed; read by the pg lane and by task db:verify, never linked
package hack/analyzers/simulable          # the INV-01 analyzer, run by task lint:simulable as its own program
package hack/analyzers/simulable/cmd/simulable # its main; `go run` builds it, no shipped binary imports it

# internal/simio/network is INV-01's fourth interface and the only one with no production
# consumer, which the real implementation used to hide: `real.NewNetwork` existed, so the
# package looked wired, and its only caller was the contract test. The real implementation
# was deleted 2026-08-09; the interface and its sim are kept because the DST harness drives
# a partition scenario through them (internal/dst/scenarios.go), which is a real assertion.
#
# Why nothing in production uses it: vhost speaks a length-delimited protocol with
# file-descriptor passing over a Unix socket, which this interface cannot express, so it
# has its own transport in internal/vhost/hostio; and the Control Plane speaks Connect over
# net/http. If a component ever needs a simulable network, this is where it plugs in.
package internal/simio/network  # harness-only: the DST partition scenario drives it, no binary does
